% Headrule: gray, full-width swelled rule with a slight taper, replacing
% fancyhdr's plain black line. Suppressed when \headrulewidth is 0 (plain
% style on chapter opening pages). The plain pagestyle sets
% \headrulewidth to 0pt, so no conditional is needed here.
% \nointerlineskip kills the
% \baselineskip glue TeX would insert before the tikz hbox (a rule node
% gets none), which would overflow the head by ~1ex.
\renewcommand{\headrule}{%
	\ifdim\headrulewidth>0pt
		\nointerlineskip
		{\color{glossgray}%
		\tikz \filldraw[scale=.04,very thin,draw=glossgray,fill=glossgray]%
		(0,0) -- (100,0.25) -- (200,0.25) -- (300,0) --%
		(200,-0.25) -- (100,-0.25) -- cycle;}%
		\vskip-\headrulewidth
	\fi
}
